cmake_minimum_required(VERSION 3.16)
project(StormMixNative VERSION 0.6.39 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Quick QuickControls2)
find_package(ALSA QUIET)
qt_standard_project_setup()
qt_add_executable(storm-mix main.cpp mixercontroller.cpp bcf2000.cpp mixercontroller.h audioengine.cpp audioengine.h dsp.h)
qt_add_qml_module(storm-mix URI StormMix VERSION 1.0 QML_FILES Main.qml)
target_link_libraries(storm-mix PRIVATE Qt6::Quick Qt6::QuickControls2)
if(ALSA_FOUND)
 target_link_libraries(storm-mix PRIVATE ALSA::ALSA)
 target_compile_definitions(storm-mix PRIVATE STORM_HAS_ALSA=1)
else()
 message(WARNING "ALSA dev library not found; build supports UI but not audio")
 target_compile_definitions(storm-mix PRIVATE STORM_HAS_ALSA=0)
endif()
install(TARGETS storm-mix RUNTIME DESTINATION bin)
add_executable(storm-dsp-test dsp_test.cpp dsp.h)
enable_testing()
add_test(NAME dsp-test COMMAND storm-dsp-test)

add_executable(storm-fft-test fft_test.cpp)
add_test(NAME fft-test COMMAND storm-fft-test)
